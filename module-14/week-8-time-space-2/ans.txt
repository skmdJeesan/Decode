Q: Calculate the time complexity for the following code snippet.
1. code - 
    for(int i = 0; i < n; i++) {
        for(int j = 0; j * j < n; j++) {
            cout << “PhysicsWallah ”;
        }
    }
    
time = O(n * sqrt(n))

2. code -
    int c = 0;
    for(int i = 0; i < n; i++) {
        for(int j = 1; j < n; j *= 2) {
            c++;
        }
    }

time = O(n * log(n))

3. code -
    int c = 0;
    for(int i = 0; i < n; i++) {
        for(int j = 1; j * j < n; j *= 2) {
            c++;
        }
    }

time = O(n * log(sqrt(n))) = O(n * log(n))

4. code -
    int c = 0;
    for(int i = n; i > 0; i /= 2) {
        for(int j = 0; j < i; j ++) {
            c++;
        }
    }

time = O(n) = O(n + n/2 + n/4 + ... + 1)

5. code -
    int c = 0;
    for(int i = 1; i < n; i*=2) {
        for(int j = n; j > i; j--) {
            c++;
        }
    }

time = O(n * log(n))
